import numpy as np
import random

class Bacterium:
    def __init__(self, energy=10.0, uptake=1.0, channel_cost=0.4, cell_id=0):
        self.energy = energy
        self.uptake = uptake  # This will now depend on channels only
        self.channel_cost = channel_cost  # Used only for building
        self.base_metabolic_cost = 0.1
        self.division_threshold = 5.0
        self.channels = 1
        self.id = cell_id

    def consume(self, nutrient):
        # Uptake depends only on number of channels and local nutrient
        uptake_per_channel = 0.2
        uptake_amount = self.channels * uptake_per_channel * nutrient
        self.energy += uptake_amount

        # Apply energy losses
        self.energy -= self.base_metabolic_cost
        self.energy -= 0.05 * self.channels  # maintenance cost (lower than base metabolic)

        # Channel building logic
        prob = self.build_channel_probability(nutrient)
        if np.random.rand() < prob:
            self.channels += 1
            self.energy -= self.channel_cost  # cost only for building

    def build_channel_probability(self, nutrient):
        if self.channels == 0:
            return 1.0  # Always build the first channel

        # Nutrient-driven building, with soft suppression
        nutrient_drive = 1 - np.exp(-3.0 * nutrient)
        suppression = self.channels / (self.channels + 100)
        probability = nutrient_drive * (1 - suppression)
        return max(0.0, min(1.0, probability))

    def ready_to_divide(self):
        return self.energy >= self.division_threshold

    def divide(self, new_id):
        energy_ratio = np.random.uniform(0.3, 0.7)
        channel_ratio = np.random.uniform(0.3, 0.7)
        child_energy = self.energy * energy_ratio
        child_channels = int(self.channels * channel_ratio)
        self.energy *= (1 - energy_ratio)
        self.channels -= child_channels

        return Bacterium(
            energy=child_energy,
            uptake=0.0,  # will be recalculated dynamically from channels
            channel_cost=self.channel_cost,
            cell_id=new_id
        )
