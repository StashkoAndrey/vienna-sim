import numpy as np
from bacteria_s2 import Bacterium
import random

class SimulationGrid:
    def __init__(self, size, channel_cost, initial_nutrient=4.0):
        self.size = size
        self.grid = [[None for _ in range(size)] for _ in range(size)]
        self.nutrients = np.full((size, size), fill_value=initial_nutrient, dtype=np.float32)
        self.channel_cost = channel_cost
        self.cell_id_counter = 0
        self.channel_time_series = {}  # Tracks number of channels over time

    def place_bacterium(self, i, j, start_energy=3.0, start_uptake=1.0):
        if self.grid[i][j] is None:
            self.grid[i][j] = Bacterium(
                energy=start_energy,
                uptake=start_uptake,
                channel_cost=self.channel_cost,
                cell_id=self.cell_id_counter
            )
            self.cell_id_counter += 1

    def step(self, timestep):
        self.channel_time_series[timestep] = []
        for i in range(self.size):
            for j in range(self.size):
                bacterium = self.grid[i][j]
                if bacterium:
                    nutrient = self.nutrients[i, j]
                    bacterium.consume(nutrient)
                    self.nutrients[i, j] = max(0.0, self.nutrients[i, j] - nutrient * bacterium.uptake)
                    self.channel_time_series[timestep].append(bacterium.channels)
                    if bacterium.energy <= 0:
                        self.grid[i][j] = None
                        continue
                    if bacterium.ready_to_divide():
                        neighbors = self.get_empty_neighbors(i, j)
                        if neighbors:
                            ni, nj = random.choice(neighbors)
                            offspring = bacterium.divide(self.cell_id_counter)
                            self.cell_id_counter += 1
                            self.grid[ni][nj] = offspring

    def refill_nutrients(self, target_total):
        current_total = np.sum(self.nutrients)
        delta = (target_total - current_total) / self.nutrients.size
        self.nutrients += delta

    def get_empty_neighbors(self, i, j):
        neighbors = []
        for di, dj in [(-1,0), (1,0), (0,-1), (0,1)]:
            ni, nj = i+di, j+dj
            if 0 <= ni < self.size and 0 <= nj < self.size:
                if self.grid[ni][nj] is None:
                    neighbors.append((ni, nj))
        return neighbors

    def count_bacteria(self):
        return sum(1 for row in self.grid for cell in row if cell is not None)

    def total_energy(self):
        return sum(cell.energy for row in self.grid for cell in row if cell is not None)

    def total_nutrients(self):
        return float(np.sum(self.nutrients))

    def get_channel_time_series(self):
        return self.channel_time_series
