import matplotlib.pyplot as plt
from grid import SimulationGrid
from diffusion_cpp import diffuse
import logging
import numpy as np
import csv
import argparse
import os

GRID_SIZE = 12
MAX_STEPS = 10000
REPLICATES = 10

# logging.basicConfig(level=logging.INFO, format='%(message)s', force=True)
logging.disable(logging.CRITICAL)  # Disable all logging

def run_simulation(channel_cost, init_nutrient, start_energy, start_uptake, diffusion_rate):
    grid = SimulationGrid(GRID_SIZE, channel_cost=channel_cost, initial_nutrient=init_nutrient)
    center = GRID_SIZE // 2
    grid.place_bacterium(center, center, start_energy=start_energy, start_uptake=start_uptake)

    population = []
    logging.info(f"\n=== Running simulation for channel_cost = {channel_cost} ===")

    for step in range(MAX_STEPS):
        grid.step()
        grid.nutrients = diffuse(grid.nutrients, diffusion_rate)
        count = grid.count_bacteria()
        total_energy = grid.total_energy()
        total_nutrient = grid.total_nutrients()
        logging.info(f"Step {step:03}: Count = {count}, Energy = {total_energy:.2f}, Nutrient = {total_nutrient:.2f}")
        population.append(count)
        if count == 0:
            logging.info("All bacteria extinct. Ending simulation early.\n")
            break

    return population


def pad_curves(curves, fill_value=0):
    max_len = max(len(c) for c in curves)
    return np.array([c + [fill_value] * (max_len - len(c)) for c in curves])


def main(batch_repeat=1):
    parser = argparse.ArgumentParser()
    parser.add_argument("--init_nutrient", type=float, default=2.0)
    parser.add_argument("--start_energy", type=float, default=3.0)
    parser.add_argument("--start_uptake", type=float, default=1.0)
    parser.add_argument("--diffusion_rate", type=float, default=0.05)
    parser.add_argument("--label", type=str, default="default")
    parser.add_argument("--output_dir", type=str, default=".")
    args = parser.parse_args()

    print(f"Starting simulation with: nutrient={args.init_nutrient}, energy={args.start_energy}, uptake={args.start_uptake}, diffusion_rate={args.diffusion_rate}, label={args.label}")

    costs = np.arange(0.0, 1.101, 0.005)

    for repeat in range(1, batch_repeat + 1):
        rep_label = f"{args.label}_R{repeat}"
        rep_output_dir = os.path.join(args.output_dir, rep_label)
        os.makedirs(rep_output_dir, exist_ok=True)

        mean_lifetimes = []
        std_lifetimes = []
        extinction_counts = []
        mean_peaks = []
        mean_totals = []
        mean_areas = []

        for cost in costs:
            lifetimes = []
            peaks = []
            totals = []
            areas = []
            extinctions = 0

            for _ in range(REPLICATES):
                curve = run_simulation(round(cost, 3), args.init_nutrient, args.start_energy, args.start_uptake, args.diffusion_rate)
                lifetimes.append(len(curve))
                peaks.append(max(curve))
                totals.append(sum(curve))
                areas.append(np.trapz(curve))
                if curve[-1] == 0:
                    extinctions += 1

            mean_lifetimes.append(np.mean(lifetimes))
            std_lifetimes.append(np.std(lifetimes))
            extinction_counts.append(extinctions)
            mean_peaks.append(np.mean(peaks))
            mean_totals.append(np.mean(totals))
            mean_areas.append(np.mean(areas))

        plot_path = os.path.join(rep_output_dir, f"lifetime_vs_cost_{rep_label}.png")
        csv_path = os.path.join(rep_output_dir, f"lifetime_data_{rep_label}.csv")

        plt.figure(figsize=(10, 5))
        plt.errorbar(costs, mean_lifetimes, yerr=std_lifetimes, fmt='-o', capsize=4)
        plt.xlabel('Channel Cost')
        plt.ylabel('Average Lifetime (steps)')
        plt.title(f'Bacterial Lifetime vs Channel Cost\n{rep_label}')
        plt.grid(True)
        plt.savefig(plot_path)
        plt.close()
        print(f"Plot saved as {plot_path}")

        with open(csv_path, "w", newline="") as f:
            writer = csv.writer(f)
            writer.writerow([
                "channel_cost", "mean_lifetime", "std_lifetime", "extinction_count",
                "mean_peak_population", "mean_total_population", "mean_area_under_curve"
            ])
            for i in range(len(costs)):
                writer.writerow([
                    round(costs[i], 5),
                    round(mean_lifetimes[i], 3),
                    round(std_lifetimes[i], 3),
                    extinction_counts[i],
                    round(mean_peaks[i], 3),
                    round(mean_totals[i], 3),
                    round(mean_areas[i], 3)
                ])

        print(f"CSV saved as {csv_path}")


if __name__ == '__main__':
    main(batch_repeat=1)
